Section 9 · Lesson 52

Multimodal RAG in Code

A PDF with a short text and a bar chart: extract both, embed both with CLIP, search, and let GPT-4.1 answer from text and image. The text never describes the bars, so questions about the chart need the image.

Setup

import fitz                                   # PyMuPDF
import io, base64, os
import numpy as np
import torch
from PIL import Image
from transformers import CLIPProcessor, CLIPModel
from langchain_core.documents import Document
from langchain_core.messages import HumanMessage
from langchain.chat_models import init_chat_model
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_community.vectorstores import FAISS
from dotenv import load_dotenv

load_dotenv()
os.environ["OPENAI_API_KEY"] = os.getenv("OPENAI_API_KEY")   # for GPT-4.1, not for CLIP

PDFs are read with PyMuPDF (fitz), which is fast and good at extracting both text and images. CLIP runs on PyTorch; the OpenAI key is for GPT-4.1.

# Initialize CLIP for unified embeddings
clip_model = CLIPModel.from_pretrained("openai/clip-vit-base-patch32")
clip_processor = CLIPProcessor.from_pretrained("openai/clip-vit-base-patch32")
clip_model.eval()

openai/clip-vit-base-patch32 turns text and images into 512-dimension vectors; the processor converts raw input into the format the model expects.

Embedding text and images

def embed_image(image_data):
    """Embed an image using CLIP"""
    if isinstance(image_data, str):          # a file path
        image = Image.open(image_data).convert("RGB")
    else:                                    # already a PIL image
        image = image_data

    inputs = clip_processor(images=image, return_tensors="pt")
    with torch.no_grad():
        features = clip_model.get_image_features(**inputs)
        features = features / features.norm(dim=-1, keepdim=True)   # unit vector
        return features.squeeze().numpy()


def embed_text(text):
    """Embed text using CLIP"""
    inputs = clip_processor(
        text=text,
        return_tensors="pt",
        padding=True,
        truncation=True,
        max_length=77,
    )
    with torch.no_grad():
        features = clip_model.get_text_features(**inputs)
        features = features / features.norm(dim=-1, keepdim=True)
        return features.squeeze().numpy()

One model, two entry points (get_image_features, get_text_features). Each vector is normalised to length 1, so text and image vectors are directly comparable.

CLIP reads only 77 tokens

Longer text is cut off before embedding, so keep text chunks short.

Processing the PDF

pdf_path = "multimodal_sample.pdf"
doc = fitz.open(pdf_path)

all_docs = []
all_embeddings = []
image_data_store = {}        # image_id -> base64 string

splitter = RecursiveCharacterTextSplitter(chunk_size=..., chunk_overlap=...)

for i, page in enumerate(doc):
    # --- Process text ---
    text = page.get_text()
    if text.strip():
        temp_doc = Document(page_content=text, metadata={"page": i, "type": "text"})
        text_chunks = splitter.split_documents([temp_doc])
        for chunk in text_chunks:
            embedding = embed_text(chunk.page_content)
            all_embeddings.append(embedding)
            all_docs.append(chunk)

    # --- Process images ---
    for img_index, img in enumerate(page.get_images(full=True)):
        try:
            xref = img[0]
            base_image = doc.extract_image(xref)
            image_bytes = base_image["image"]

            # 1. Convert to a PIL image
            pil_image = Image.open(io.BytesIO(image_bytes)).convert("RGB")
            image_id = f"page_{i}_img_{img_index}"

            # 2. Store as base64 for the multimodal LLM
            buffered = io.BytesIO()
            pil_image.save(buffered, format="PNG")
            img_base64 = base64.b64encode(buffered.getvalue()).decode()
            image_data_store[image_id] = img_base64

            # 3. Create a CLIP embedding for retrieval
            embedding = embed_image(pil_image)
            all_embeddings.append(embedding)

            image_doc = Document(
                page_content=f"[Image: {image_id}]",
                metadata={"page": i, "type": "image", "image_id": image_id},
            )
            all_docs.append(image_doc)
        except Exception as e:
            print(f"Error processing image {img_index} on page {i}: {e}")

doc.close()

Text, per page

Get the text, wrap it in a Document tagged "type": "text", chunk it, embed each chunk.

Images, per page

Extract each image, store it as base64 under an ID, embed it, and add a placeholder Document tagged "type": "image" with that ID.

Result: 2 Documents (one text, one image) and their two vectors. The type tag is how they're told apart later.

Storing and searching

embeddings_array = np.array(all_embeddings)

vector_store = FAISS.from_embeddings(
    text_embeddings=[(doc.page_content, emb) for doc, emb in zip(all_docs, embeddings_array)],
    embedding=None,                                  # embeddings already computed
    metadatas=[doc.metadata for doc in all_docs],
)

The vectors were already made by CLIP, so from_embeddings stores them as-is (no embedding model in the store).

llm = init_chat_model("openai:gpt-4.1")      # a multimodal LLM

def retrieve_multimodal(query, k=5):
    """Unified retrieval using CLIP embeddings for both text and images"""
    query_embedding = embed_text(query)
    results = vector_store.similarity_search_by_vector(embedding=query_embedding, k=k)
    return results

For the same reason, search uses similarity_search_by_vector with the CLIP-embedded query; the results can be text or images.

The message for the LLM

def create_multimodal_message(query, retrieved_docs):
    """Create a message with both text and images for GPT-4.1"""
    content = []
    content.append({"type": "text", "text": f"Question: {query}\n\nContext:\n"})

    # Separate text and image documents
    text_docs = [d for d in retrieved_docs if d.metadata.get("type") == "text"]
    image_docs = [d for d in retrieved_docs if d.metadata.get("type") == "image"]

    if text_docs:
        text_context = "\n\n".join(
            f"[Page {d.metadata['page']}]: {d.page_content}" for d in text_docs
        )
        content.append({"type": "text", "text": f"Text excerpts:\n{text_context}\n"})

    for d in image_docs:
        image_id = d.metadata.get("image_id")
        if image_id and image_id in image_data_store:
            content.append({"type": "text", "text": f"\n[Image from page {d.metadata['page']}]:\n"})
            content.append({
                "type": "image_url",
                "image_url": {"url": f"data:image/png;base64,{image_data_store[image_id]}"},
            })

    content.append({"type": "text", "text": "\n\nPlease answer the question based on the provided text and images."})
    return HumanMessage(content=content)

A multimodal LLM takes one message made of parts: text parts for the question and text chunks, and image_url parts holding each retrieved image as a base64 data URL. That's why the images were kept as base64.

Results

def multimodal_pdf_rag_pipeline(query):
    context_docs = retrieve_multimodal(query, k=5)
    message = create_multimodal_message(query, context_docs)
    response = llm.invoke([message])

    print(f"Retrieved {len(context_docs)} documents:")
    for d in context_docs:
        kind = d.metadata.get("type")
        print(f"- {kind} from page {d.metadata['page']}")
    return response.content

if __name__ == "__main__":
    queries = [
        "What does the chart on page 1 show about revenue trends?",
        "Summarize the main findings from the document",

Takeaways

  • PyMuPDF extracts text and images; CLIP embeds both into one space.
  • Tag Documents by type and keep images as base64.
  • Precomputed vectors: from_embeddings to store, similarity_search_by_vector to search.
  • Send text and image parts together to a multimodal LLM.
